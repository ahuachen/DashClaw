# Hosted Workspace Provisioning Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a gated hosted-mode provisioning endpoint that atomically creates a trial org + API key, plus middleware that enforces trial expiry and action caps. This is Plan 1 of 4 in the "click to try hosted DashClaw" initiative.

**Architecture:** Extend the existing `organizations` and `api_keys` tables with hosted-trial columns. Add `/api/hosted/workspaces` routes that are 404 unless `DASHCLAW_HOSTED=true`. Provisioning is protected by Cloudflare Turnstile + in-memory IP rate limiting. Trial enforcement lives in `middleware.js` alongside existing API-key resolution so every governed call checks expiry/cap at no extra cost. Self-host deployments are unaffected because the flag defaults off.

**Tech Stack:** Next.js 16 App Router, Drizzle ORM, Neon Postgres (HTTP driver), Vitest + jsdom, Cloudflare Turnstile (server-side verify via `https://challenges.cloudflare.com/turnstile/v0/siteverify`).

**Scope boundary:**
- IN: new columns on `organizations`, `/api/hosted/workspaces` routes, middleware enforcement, tests, docs.
- OUT: stack picker UI (Plan 2), installer templates (Plan 3), hosted-instance infra (Plan 4), Stripe upgrade flow, email capture.

**Non-negotiable constraints (from project memory):**
- No LLM API keys in this flow — the only credential minted is a DashClaw workspace API key.
- Follow the "Route SQL guardrail" — no inline SQL in route files; use repositories in `app/lib/repositories/`.
- All SDK docs sites must be updated when adding API routes (see `MEMORY.md` SDK Documentation Checklist).
- `graphify-pilot/` is a snapshot — never edit; real schema is `schema/schema.js`.

```javascript
const SITEVERIFY_URL = 'https://challenges.cloudflare.com/turnstile/v0/siteverify';

export async function verifyTurnstile(token, remoteIp) {
  const secret = process.env.TURNSTILE_SECRET_KEY;
  if (!secret) {
    return { ok: true, bypassed: true };
  }
  if (!token) {
    return { ok: false, reason: 'missing_token' };
  }
  try {
    const body = new URLSearchParams();
    body.set('secret', secret);
    body.set('response', token);
    if (remoteIp) body.set('remoteip', remoteIp);
    const res = await fetch(SITEVERIFY_URL, {
      method: 'POST',
      headers: { 'content-type': 'application/x-www-form-urlencoded' },
      body,
    });
    const json = await res.json();
    if (json.success) return { ok: true };
    return { ok: false, reason: 'cf_rejected', errors: json['error-codes'] || [] };
  } catch (err) {
    console.error('[HOSTED] turnstile verify failed:', err.message);
    return { ok: false, reason: 'verify_failed' };
  }
}
```

```javascript
export function createRateLimiter({ max, windowMs }) {
  const hits = new Map();

  function prune(now) {
    for (const [key, entry] of hits) {
      if (entry.resetAt <= now) hits.delete(key);
    }
  }

  return {
    take(ip) {
      if (!ip) return { ok: true, remaining: max };
      const now = Date.now();
      prune(now);
      const entry = hits.get(ip) ?? { count: 0, resetAt: now + windowMs };
      if (entry.count >= max) {
        return { ok: false, retryAfterMs: Math.max(0, entry.resetAt - now) };
      }
      entry.count += 1;
      hits.set(ip, entry);
      return { ok: true, remaining: max - entry.count };
    },
  };
}
```

```javascript
import { neon } from '@neondatabase/serverless';
import crypto from 'node:crypto';

function getSql() {
  return neon(process.env.DATABASE_URL);
}

function generateId(prefix) {
  return `${prefix}_${crypto.randomBytes(12).toString('hex')}`;
}

function generateApiKey() {
  const raw = crypto.randomBytes(16).toString('hex');
  const plaintext = `oc_live_${raw}`;
  const keyHash = crypto.createHash('sha256').update(plaintext).digest('hex');
  const keyPrefix = plaintext.slice(0, 12);
  return { plaintext, keyHash, keyPrefix };

export async function deleteHostedWorkspace(orgId) {

    SELECT hosted_mode FROM organizations WHERE id = ${orgId} LIMIT 1
  `;
```

```javascript
import { NextResponse } from 'next/server';
import { isHostedMode, hostedConfig } from '../../../lib/hosted/flag.js';
import { verifyTurnstile } from '../../../lib/hosted/turnstile.js';
import { createRateLimiter } from '../../../lib/hosted/rate-limit.js';
import { provisionHostedWorkspace } from '../../../lib/repositories/hosted-workspace.repository.js';

const DAY_MS = 24 * 60 * 60 * 1000;
let _limiter = null;
function getLimiter() {
  const cfg = hostedConfig();
  if (!_limiter || _limiter._max !== cfg.maxProvisionsPerIpPerDay) {
    _limiter = createRateLimiter({ max: cfg.maxProvisionsPerIpPerDay, windowMs: DAY_MS });
    _limiter._max = cfg.maxProvisionsPerIpPerDay;
  }
  return _limiter;
}

function clientIp(request) {
  const fwd = request.headers.get('x-forwarded-for');
  if (fwd) return fwd.split(',')[0].trim();
  return request.headers.get('x-real-ip') || null;
}

function publicEndpoint(request) {
  const url = new URL(request.url);
  return `${url.protocol}//${url.host}`;
}

export async function POST(request) {
  if (!isHostedMode()) {
    return NextResponse.json({ error: 'Not found' }, { status: 404 });
  }

  const ip = clientIp(request);
  const rl = getLimiter().take(ip);
  if (!rl.ok) {
    return NextResponse.json(
      { error: 'Rate limit exceeded', retry_after_ms: rl.retryAfterMs },
      { status: 429 },
    );
  }

  let body = {};
  try { body = await request.json(); } catch { /* empty body allowed */ }

  const turnstile = await verifyTurnstile(body.turnstile_token || '', ip);
  if (!turnstile.ok) {
    return NextResponse.json(
      { error: `turnstile verification failed: ${turnstile.reason}` },
      { status: 400 },
    );
  }

  const cfg = hostedConfig();
  try {
    const result = await provisionHostedWorkspace({
      trialDays: cfg.trialDays,
      trialActionCap: cfg.trialActionCap,
      label: 'trial',
    });
    return NextResponse.json({
      workspace_id: result.orgId,
      api_key: result.apiKey,
      key_prefix: result.keyPrefix,
      endpoint: publicEndpoint(request),
      expires_at: result.expiresAt,
      trial_action_cap: cfg.trialActionCap,
      next_steps_url: `${publicEndpoint(request)}/connect?hosted=${result.orgId}`,
    });
  } catch (err) {
    console.error('[HOSTED] provision failed:', err);
    return NextResponse.json({ error: 'Provisioning failed' }, { status: 500 });
  }
}

export async function GET() {
  return NextResponse.json({ error: 'Method not allowed' }, { status: 405 });
```

```javascript
export async function GET(request, { params }) {
  if (!isHostedMode()) return NextResponse.json({ error: 'Not found' }, { status: 404 });
  if (!requireAdmin(request)) return NextResponse.json({ error: 'Forbidden' }, { status: 403 });

    trial_action_cap: ws.trialActionCap,
    trial_actions_used: ws.trialActionsUsed,

export async function DELETE(request, { params }) {
  if (!isHostedMode()) return NextResponse.json({ error: 'Not found' }, { status: 404 });
  if (!requireAdmin(request)) return NextResponse.json({ error: 'Forbidden' }, { status: 403 });

  const { workspaceId } = await params;
```

```javascript
import { NextResponse } from 'next/server';
import { isHostedMode } from '../../../lib/hosted/flag.js';
import { findExpiredWorkspaces, deleteHostedWorkspace } from '../../../lib/repositories/hosted-workspace.repository.js';

function requireAdmin(request) {
  const role = request.headers.get('x-role');
  if (role === 'owner' || role === 'admin') return true;
  // Also accept a shared secret for cron jobs
  const secret = request.headers.get('x-cleanup-secret');
  return !!(secret && process.env.HOSTED_CLEANUP_SECRET && secret === process.env.HOSTED_CLEANUP_SECRET);
}

export async function POST(request) {
  if (!isHostedMode()) return NextResponse.json({ error: 'Not found' }, { status: 404 });
  if (!requireAdmin(request)) return NextResponse.json({ error: 'Forbidden' }, { status: 403 });

  const expired = await findExpiredWorkspaces({ now: new Date(), limit: 100 });
  let deleted = 0;
  const errors = [];
  for (const orgId of expired) {
    try {
      const r = await deleteHostedWorkspace(orgId);
      if (r.deleted) deleted += 1;
    } catch (err) {
      errors.push({ orgId, error: err.message });
    }
  }
  return NextResponse.json({ found: expired.length, deleted, errors });
}
```
