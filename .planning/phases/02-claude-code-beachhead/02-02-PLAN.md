    ```javascript
    export const dynamic = 'force-dynamic';
    export const revalidate = 0;

    import { NextResponse, after } from 'next/server';
    import nacl from 'tweetnacl';
    import { getSql } from '../../../lib/db.js';
    import {
      getActionSummary,
      recordApproval,
    } from '../../../lib/repositories/actions.repository.js';

    const DISCORD_API = 'https://discord.com/api/v10';
    const CALLBACK_DATA_RE = /^(ap|dn):(act_[a-z0-9_-]{1,57})$/;
    const FETCH_TIMEOUT_MS = 1500;
    // Replay window: reject timestamps older than 5 minutes (T-02-02-02 mitigation)
    const TIMESTAMP_SKEW_SECONDS = 5 * 60;

    const PING = 1;
    const MESSAGE_COMPONENT = 3;
    const PONG = 1;
    const DEFERRED_UPDATE_MESSAGE = 6;

    function verifyDiscordSignature(rawBody, signatureHex, timestampStr, publicKeyHex) {
      if (!signatureHex || !timestampStr || !publicKeyHex) return false;
      // T-02-02-02: reject stale timestamps (anti-replay)
      const now = Math.floor(Date.now() / 1000);
      const ts = Number(timestampStr);
      if (!Number.isFinite(ts) || Math.abs(now - ts) > TIMESTAMP_SKEW_SECONDS) return false;
      try {
        return nacl.sign.detached.verify(
          Buffer.from(timestampStr + rawBody),
          Buffer.from(signatureHex, 'hex'),
          Buffer.from(publicKeyHex, 'hex'),
        );
      } catch {
        return false;
      }
    }

    export async function POST(request) {
      // Verify BEFORE JSON.parse (Pitfall 1 — re-serialization breaks signature)
      const rawBody = await request.text();
      const sig = request.headers.get('x-signature-ed25519');
      const ts = request.headers.get('x-signature-timestamp');

      if (!verifyDiscordSignature(rawBody, sig, ts, process.env.DISCORD_PUBLIC_KEY)) {
        // T-02-02-01: MUST return 401 (never 200/403) — Discord security sweeps
        return NextResponse.json({ error: 'invalid request signature' }, { status: 401 });
      }

      let body;
      try { body = JSON.parse(rawBody); } catch {
        return NextResponse.json({ error: 'Invalid JSON' }, { status: 400 });
      }

      // PING handshake — Discord Developer Portal saves interactions URL only
      // if this returns {type: 1}
      if (body.type === PING) return NextResponse.json({ type: PONG });

      if (body.type !== MESSAGE_COMPONENT) return NextResponse.json({ type: PONG });

      // T-02-02-03: user identity check AFTER signature verify.
      // DM context uses body.user.id; guild context uses body.member.user.id.
      // Phase 2 is DM-only per D-04, so prefer body.user.id.
      const userId = String(body.user?.id ?? body.member?.user?.id ?? '');
      if (userId !== process.env.DISCORD_APPROVER_USER_ID) {
        return NextResponse.json({ error: 'invalid request signature' }, { status: 401 });
      }

      const customId = body.data?.custom_id ?? '';
      const match = customId.match(CALLBACK_DATA_RE);
      if (!match) return NextResponse.json({ type: DEFERRED_UPDATE_MESSAGE });
      const [, verb, actionId] = match;

      const appId = body.application_id;
      const interactionToken = body.token;

      // Ack inside the 3-second window; DB work happens in `after()`.
      after(() => resolveApproval(verb, actionId, userId, appId, interactionToken));

      return NextResponse.json({ type: DEFERRED_UPDATE_MESSAGE });
    }

    async function resolveApproval(verb, actionId, discordUserId, appId, interactionToken) {
      const orgId = process.env.DISCORD_APPROVER_ORG_ID;
      if (!orgId) {
        await editOriginal(appId, interactionToken,
          'Server misconfigured: DISCORD_APPROVER_ORG_ID is not set');
        return;
      }
      const sql = getSql();
      const action = await getActionSummary(sql, orgId, actionId);
      if (!action) { await editOriginal(appId, interactionToken, 'Action not found'); return; }
      if (action.status !== 'pending_approval') {
        await editOriginal(appId, interactionToken, `Already resolved — status: ${action.status}`);
        return;
      }
      const userId = `discord:${discordUserId}`;
      const isApprove = verb === 'ap';
      let updated;
      try {
        updated = await recordApproval(sql, orgId, actionId, {
          newStatus: isApprove ? 'running' : 'failed',
          errorMessage: isApprove ? null : 'Denied via Discord',
          decision: isApprove ? 'allow' : 'deny',
          userId,
          safeReasoning: isApprove ? null : 'Denied via Discord',
        });
      } catch (err) {
        console.warn('[DiscordInteractions] recordApproval failed:', err.message);
        await editOriginal(appId, interactionToken, 'Approval failed');
        return;
      }
      if (!updated) {
        await editOriginal(appId, interactionToken, 'Already resolved — resolved by another channel');
        return;
      }
      await editOriginal(appId, interactionToken,
        buildResolvedText(action, isApprove ? 'APPROVED' : 'DENIED', actionId));
    }

    async function editOriginal(appId, interactionToken, content) {
      try {
        await fetch(`${DISCORD_API}/webhooks/${appId}/${interactionToken}/messages/@original`, {
          method: 'PATCH',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ content, components: [] }),
          signal: AbortSignal.timeout(FETCH_TIMEOUT_MS),
        });
      } catch (err) {
        console.warn('[DiscordInteractions] editOriginal failed:', err.message);
      }
    }

    function buildResolvedText(action, decisionLabel, actionId) {
      const ts = new Date().toTimeString().slice(0, 8);
      const goal = (action.declared_goal || '—').slice(0, 200);
      return [
        `${decisionLabel} — ${ts}`, '',
        `Agent:   ${action.agent_id || 'unknown'}`,
        `Action:  ${action.action_type || 'unknown'}`,
        `Goal: ${goal}`, '',
        actionId,
      ].join('\n');
    }
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Discord → /api/discord/interactions | Untrusted: anyone can POST to this endpoint. Discord's Ed25519 signature is the only thing distinguishing legitimate interactions from attackers. |
| DashClaw server → discord.com/api/v10 | Trusted outbound (our bot token authenticates as us); TLS via Vercel. |
| Hook → /api/actions → fireDiscordApproval | Trusted in-process call chain, same org-scoped authentication path Telegram already uses. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-02-01 | Spoofing | `POST /api/discord/interactions` — fake interactions without valid Ed25519 signature | mitigate | `verifyDiscordSignature` runs on raw bytes (`await request.text()` — NOT `request.json()`) BEFORE JSON.parse; returns 401 on any mismatch. Asserted by `discord-interactions-route.test.js` tests 1-3. Discord's automated security sweeps will disable the Interactions URL if this returns anything but 401 on bad signatures (RESEARCH §Pitfall 3). |
| T-02-02-02 | Tampering | Replay of captured signed interaction payload | mitigate | `TIMESTAMP_SKEW_SECONDS = 5 * 60` — reject any request where `abs(now - headers['x-signature-timestamp']) > 300`. Discord includes the timestamp in the signed data so an attacker can't tamper with it without breaking the signature. Implemented in `verifyDiscordSignature` before the nacl.verify call. |
| T-02-02-03 | Spoofing | custom_id injection — attacker crafts button callback with malicious action_id | mitigate | `CALLBACK_DATA_RE = /^(ap\|dn):(act_[a-z0-9_-]{1,57})$/` strict regex. Repository `getActionSummary(sql, orgId, actionId)` scopes by `DISCORD_APPROVER_ORG_ID` — an action_id from a different org returns null, route responds with "Action not found" via PATCH @original and never calls recordApproval. Asserted by test 11 (misconfig) + idempotency tests 9-10. |
| T-02-02-04 | Spoofing | Different Discord user impersonates the approver | mitigate | `body.user.id === DISCORD_APPROVER_USER_ID` check AFTER signature verify passes. 401 on mismatch (not 403 — avoids leaking "signature correct but sender wrong" distinguishability). Asserted by test 4. |
| T-02-02-05 | Information Disclosure | Bot token or signature headers leaked via logs | mitigate | `console.warn` sites log only `err.message` (never the raw headers or token). Mirrors Telegram discipline at `telegram/webhook/route.js:33,49`. No new log statements introduce env-var readback. |
| T-02-02-06 | Denial of Service | Attacker floods `/api/discord/interactions` with malformed bodies | accept | Signature verify is the first operation; bad-sig requests cost ~100µs to reject. JSON.parse failures die at 400 before touching repos. Vercel's per-lambda concurrency cap + Discord's own rate limits cap realistic damage. Low-value target (no PII, no state mutation without valid sig + correct user_id). |
| T-02-02-07 | Tampering | Race between Telegram + Discord approvals on the same action | mitigate | Repository `recordApproval` uses atomic `UPDATE WHERE status='pending_approval' RETURNING *`.
